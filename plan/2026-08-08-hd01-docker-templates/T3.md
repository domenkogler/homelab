**Exact template content:**

# loki docker-compose.yml — single-node log aggregation (14d retention, SSD)
#   Alloy (host agent) pushes container/host logs here on db-internal.
#   Grafana queries Loki over db-internal; no web UI of its own, no Traefik labels.
#   TSDB data is regenerable and NOT backed up (docs/backup.md).
#   See docs/observability.md. Secrets via
#   {{ lookup('community.general.onepassword', '<name>', vault=op_vault) }} — none needed here.
services:
  loki:
    image: grafana/loki:latest
    container_name: loki
    restart: unless-stopped
    command:
      - "-config.file=/etc/loki/loki.yaml"
      - "-config.expand-env=true"
    networks:
      - db-internal
    volumes:
      - ./loki.yaml:/etc/loki/loki.yaml:ro
      - loki-data:/loki
    ports:
      - "3100:3100"        # HTTP API — consumed by Grafana/Alloy on db-internal; not exposed publicly
volumes:
  loki-data:
networks:
  db-internal:
    external: true
```
`loki.yaml.j2` exact content:

# Loki single-node config — 14d retention, filesystem/TSDB store (docs/observability.md)
auth_enabled: false
server:
  http_listen_port: 3100
common:
  path_prefix: /loki
  storage:
    filesystem:
      chunks_directory: /loki/chunks
      rules_directory: /loki/rules
  replication_factor: 1
  ring:
    kvstore:
      store: inmemory
schema_config:
  configs:
    - from: 2026-01-01
      store: tsdb
      object_store: filesystem
      schema: v13
      index:
        prefix: index_
        period: 24h
limits_config:
  retention_period: 14d
  retention_stream:
    - selector: '{namespace="default"}'
      priority: 1
      period: 14d
compactor:
  working_directory: /loki/compactor
  compaction_interval: 5m
table_manager:
  retention_deletes_enabled: true
  retention_period: 14d
```
