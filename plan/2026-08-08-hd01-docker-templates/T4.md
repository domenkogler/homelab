#   service-internal (needs internet). Linked to Domen's personal number (no second SIM),
#   sends to the dedicated "Homelab Alerts" group.

#   NOT acceptable (docs/observability.md). Secrets (signal_api_*) via
#   {{ lookup('community.general.onepassword', '<name>', vault=op_vault) }}
services:
  signal-cli-rest-api:
    image: bbernhard/signal-cli-rest-api:latest
    container_name: signal-cli-rest-api
    restart: unless-stopped
    environment:
      SIGNAL_CLI_PHONE_NUMBER: "{{ lookup('community.general.onepassword', 'signal_api_phone_number', vault=op_vault) }}"
      SIGNAL_CLI_CAPTCHA: "{{ lookup('community.general.onepassword', 'signal_api_captcha', vault=op_vault) | default('', true) }}"
      MODE: json-rpc
    networks:
      - services-internal
    volumes:
      - signal-cli-data:/home/.local/share/signal-cli      # identity persists across restarts
    ports:
      - "8080:8080"        # HTTP API — consumed by n8n on services-internal; not exposed publicly
volumes:
  signal-cli-data:
networks:
  services-internal:
    external: true
